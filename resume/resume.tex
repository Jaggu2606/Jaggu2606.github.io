%-----------RESEARCH-----------------
\section{Research Experience -- Robotics}
\begin{entries}
  \project{AdaTerrainDiff: Terrain-Conditioned Adaptive Action Refinement for Bipedal Locomotion}{2026 -- Ongoing}
          {Under \textbf{Dr. Amit Shukla}, Indian Institute of Technology Mandi}
  \begin{bullets}
    \item Diffusion locomotion policy with a disentangled \textbf{128-D state latent} and \textbf{16-D terrain latent}; the terrain latent sets the number of denoising steps (5 on flat ground, up to 10 on stairs). Trained with online \textbf{PPO} and an $x_0$-prediction denoiser.
    \item One policy for flat ground, slopes, rough terrain, discrete obstacles and stairs: \textbf{81.0\%} average success vs. 78.0\% (BiRoDiff) and 32.1\% (DiffuseLoco).
    \item Sim-to-real on the 10-DOF biped completed \textbf{98.3\%} of each 60\,s episode, with cost of transport within 13\% of simulation.
  \end{bullets}

  \project{Terrain-Adaptive Locomotion for a 10-DOF Biped with Shared Autonomy}{2025 -- 2026}
          {Under \textbf{Dr. Amit Shukla}, Indian Institute of Technology Mandi}
  \begin{bullets}
    \item Built a \textbf{10-DOF biped} (based on the Duke Humanoid) for industrial inspection sites; RL with structured terrain encoding for flat ground, stairs and uneven terrain, trained in \textbf{Isaac Gym} with domain randomization.
    \item Low-level safety loop for stability margins and joint compliance; custom \textbf{BLDC actuator} control and PID tuning; gantry-supported sim-to-real and \textbf{push-recovery} trials.
  \end{bullets}

  \project{Multi-Skill Bipedal Robot with Learning-Based Control (M.Tech Thesis)}{Mar 2024 -- Sep 2026}
          {Under \textbf{Dr. Amit Shukla}, Indian Institute of Technology Mandi}
  \begin{bullets}
    \item Designed a small-scale biped from scratch: mechanical design, kinematic modeling and stability verification.
    \item \textbf{PPO, TD3 and SAC} policies with waypoint-based reward shaping reach \textbf{over 90\%} success in walking and running; developed in Gazebo, PyBullet and Isaac Gym and deployed through a \textbf{ROS2 sim-to-real} pipeline.
  \end{bullets}

  \project{Hierarchical RL for Collision-Free Locomotion of an Underactuated Biped}{Feb 2026 -- Jul 2026}
          {Under \textbf{Dr. Amit Shukla}, Indian Institute of Technology Mandi}
  \begin{bullets}
    \item High-level waypoint policy plus low-level gait policy, trained in PyBullet with up to 20 obstacles: \textbf{98\%} success (SPL 0.955) in simulation and \textbf{92\%} (SPL 0.884) on hardware.
  \end{bullets}

  \project{Deep RL for Vision-Based UAV Tracking by an Autonomous Ground Vehicle}{Jan 2026 -- Aug 2026}
          {Under \textbf{Dr. Amit Shukla}, Centre for AI \& Robotics, IIT Mandi \ [\href{https://microgridx.github.io/UAV-AGV_Collaboration_website/}{Website}]}
  \begin{bullets}
    \item AGV tracks a \textbf{DJI Matrice 600} using a real-time \textbf{YOLOv8n} detector (5,000-frame custom dataset) on an upward camera; composite reward with a potential-based progress term.
    \item Benchmarked DQN, TD3, SAC, TQC and \textbf{TD7}; TD7 gave the lowest steady-state error and fastest convergence.
  \end{bullets}

  \project{Autonomous Buried Power Cable Tracing}{Jan 2025 -- Jun 2026}
          {Under \textbf{Dr. Amit Shukla}, Indian Institute of Technology Mandi}
  \begin{bullets}
    \item Ground robot that follows buried cables from \textbf{electromagnetic locator feedback}, with goal-free obstacle avoidance and \textbf{GPS mapping}; validated in ROS/Gazebo and outdoor hardware trials.
  \end{bullets}
\end{entries}
